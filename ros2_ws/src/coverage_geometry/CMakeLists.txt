cmake_minimum_required(VERSION 3.16)
project(coverage_geometry)
find_package(ament_cmake REQUIRED)
find_package(nav2_controller REQUIRED)
find_package(nav2_util REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
add_library(ordered_goal_checker SHARED src/ordered_goal_checker.cpp)
target_compile_features(ordered_goal_checker PRIVATE cxx_std_17)
ament_target_dependencies(ordered_goal_checker nav2_controller nav2_util nav_msgs pluginlib tf2_geometry_msgs)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
install(TARGETS ordered_goal_checker LIBRARY DESTINATION lib)
ament_package()
