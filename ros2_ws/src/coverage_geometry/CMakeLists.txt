cmake_minimum_required(VERSION 3.16)
project(coverage_geometry)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(Fields2Cover REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Development.Module)
find_package(nav2_controller REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
add_library(ordered_goal_checker SHARED src/ordered_goal_checker.cpp)
target_compile_features(ordered_goal_checker PRIVATE cxx_std_17)
ament_target_dependencies(ordered_goal_checker nav2_controller nav_msgs pluginlib tf2_geometry_msgs)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
install(TARGETS ordered_goal_checker LIBRARY DESTINATION lib)
Python3_add_library(_coverage_geometry MODULE src/geometry.cpp)
target_compile_features(_coverage_geometry PRIVATE cxx_std_17)
target_link_libraries(_coverage_geometry PRIVATE Fields2Cover)
ament_get_python_install_dir(PYTHON_INSTALL_DIR)
_ament_cmake_python_register_environment_hook()
install(TARGETS _coverage_geometry LIBRARY DESTINATION "${PYTHON_INSTALL_DIR}")
ament_package()
